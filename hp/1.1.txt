#include <iostream>
#include<vector>
#include<queue>
#include<omp.h>


using namespace std;

int vertices;//store number of nodes
vector<vector<int>> adjList;//adjlist=store graph connections

void addEdge(int u,int v){//adds an edge between 2 nodes
	adjList[u].push_back(v);
	adjList[v].push_back(u);
}

void parallelBFS(int start){//start=starting node
	vector<bool> visited(vertices,false);//keeps track of visited nodes ,initially no node visited

	queue<int> q;//queue works with fifo
	visited[start] = true;//mark start node as visited and add to queue
	q.push(start);//node is pushed into queue


	cout<<"BFS: ";

	while(!q.empty()){//Run untill queue becomes empty

		int levelsize = q.size();//stores how many nodes are in current level
		vector<int> currentLevel;//stores nodes of current bfs level

		for(int i = 0;i<levelsize;i++){//loop runs for all nodes of current levelx
			currentLevel.push_back(q.front());//<-
			q.pop();
		}


		//prtocess current level nodes in parallel
		//each thread handles one node from this level

		#pragma omp parallel for//run for loop in parallel using multiple threads
		for(int i = 0;i<(int)currentLevel.size();i++){//processes all nodes of current bfs level
			int node = currentLevel[i];//stroes current node

			//only one thread prints at a time
			#pragma omp critical//many threads may try to print together so tae 1 thread and print
			cout<<node<< " ";

			//check neighbours
			for(int neighbor : adjList[node]){
				#pragma omp critical//this protects shared data
				{

				if(!visited[neighbor]){
					visited[neighbor]= true;
					q.push(neighbor);

				}
				}
			}

		}
	}

	cout<<endl;


}
void parallelDFS(int start){
	vector<bool> visited(vertices,false);

	// dfs uses a stack instead of queues
	vector<int> stk;
	stk.push_back(start);//push starting node

	cout<<"DFS: ";

	while(!stk.empty()){//loop continues until stack becomes empty
		// take the top element
		int node = stk.back();//stk.back returns the last element of stack
		stk.pop_back();//removes last element

		//process only if not visited
		if(!visited[node]){//if node is not visited then process it
			visited[node] = true;//marks node as visited


			#pragma omp critical
			cout<<node<<" ";

			// push all unvisited neighbors in parallel
			#pragma omp parallel for
			for(int i = 0;i<(int)adjList[node].size();i++){ //this checks all neighbours of current node

				int neighbor = adjList[node][i];//gets current node
				#pragma omp critical
				{
					if(!visited[neighbor]){
						stk.push_back(neighbor);//add neighbor into dfs stack
					}
				}
			}
		}
	}
	cout<<endl;
}

int main(){

	int edges;

	cout<< "Enter number of vertices:";
	cin>> vertices;
	adjList.resize(vertices); //for making the adjacency list fready


	cout<< "Enter number of edges:";
	cin>>edges;


	cout<< "Enter the numbetr of edges as u v\n";


	for(int i = 0;i<edges;i++){
		int u,v;
		cin>>u>>v;
		addEdge(u,v);
	}

	int start;
	cout<<"Enter the starting node:";
	cin>>start;


	//measure the BFS time
	double t1 = omp_get_wtime();//start time before bfs
	parallelBFS(start);//executes bfs
	double t2 = omp_get_wtime();//stops time after bfs finishes
	cout<<"BFS Time:"<<(t2-t1)<<"sec\n\n";//calculate total bfs time


	//measure the DFS time
	double t3 = omp_get_wtime();
	parallelDFS(start);
	double t4 = omp_get_wtime();
	cout<< "DFS time:"<<(t4-t3) << "sec\n";

	return 0;




}
