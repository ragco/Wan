%%writefile vector_input.cu
#include <stdio.h>  //standard input output library

__global__ void vectorAdd(int *A, int *B, int *C, int n) {  //cuda keyword
    int i = blockIdx.x * blockDim.x + threadIdx.x; //global thread ID 
    if (i < n) {
        C[i] = A[i] + B[i];
    }
}

int main() {
    int N; //stores vector size
    printf("Enter size of vector: "); 
    scanf("%d", &N);

    int A[N], B[N], C[N]; //3 arrayys created in ram (cpu memory)
    int *d_A, *d_B, *d_C;  // d_ means device(GPU memory)these pointers will store address of A,B,C in GPU memory
    int size = N * sizeof(int);

    printf("Enter elements of A:\n");
    for(int i = 0; i < N; i++) scanf("%d", &A[i]);

    printf("Enter elements of B:\n");
    for(int i = 0; i < N; i++) scanf("%d", &B[i]);

    cudaMalloc((void**)&d_A, size); //allocates memory inside gpu
    cudaMalloc((void**)&d_B, size);
    cudaMalloc((void**)&d_C, size);

    cudaMemcpy(d_A, A, size, cudaMemcpyHostToDevice); // copies arrays from cpu to gpu
    cudaMemcpy(d_B, B, size, cudaMemcpyHostToDevice);

    vectorAdd<<<1, N>>>(d_A, d_B, d_C, N);  //kernel launch syntax(blocks,threads)
    cudaDeviceSynchronize();  //cpu waits until gpu finishes execution

    cudaMemcpy(C, d_C, size, cudaMemcpyDeviceToHost);  //copies result back to cpu

    printf("Result:\n");
    for(int i = 0; i < N; i++) {
        printf("%d ", C[i]);
    }

    cudaFree(d_A);
    cudaFree(d_B);
    cudaFree(d_C);

    return 0;
}


!nvcc vector_input.cu -o vector_input

!./vector_input

%%writefile matrix_input.cu
#include <stdio.h>

#define MAX 10  // max matrix sixe = 10*`10

__global__ void matrixMul(int *A, int *B, int *C, int n) {  //a 1st matrix, B 2nd matrix , C result matrix , n matrix size
    int row = threadIdx.y;  // every thread has row number and column number
    int col = threadIdx.x;

    int sum = 0;  //stores ans temporarily
    for(int k = 0; k < n; k++) {  // loop for matrix multiplication
        sum += A[row*n + k] * B[k*n + col];  // does marix multiplication
    }
    C[row*n + col] = sum;
}

int main() {
    int n;
    printf("Enter size of matrix: ");
    scanf("%d", &n);

    int A[MAX][MAX], B[MAX][MAX], C[MAX][MAX];  //matrices stored in cpu ram
    int *d_A, *d_B, *d_C;  // pointers for gpu memory

    printf("Enter Matrix A:\n");
    for(int i=0;i<n;i++)
        for(int j=0;j<n;j++)
            scanf("%d",&A[i][j]);  // reads matrix elemnts of A

    printf("Enter Matrix B:\n");
    for(int i=0;i<n;i++)
        for(int j=0;j<n;j++)
            scanf("%d",&B[i][j]);  // reads matrix elemnets of B 

    int size = n*n*sizeof(int);  // calculate memory size

    cudaMalloc((void**)&d_A, size);  // allocates gpu memory
    cudaMalloc((void**)&d_B, size);
    cudaMalloc((void**)&d_C, size);

    cudaMemcpy(d_A, A, size, cudaMemcpyHostToDevice);  //copy cpu to gpu
    cudaMemcpy(d_B, B, size, cudaMemcpyHostToDevice);

    dim3 threads(n,n);  //creates 2D thread block
    matrixMul<<<1,threads>>>(d_A,d_B,d_C,n);  // kernel launch(blocks,threads)
    cudaDeviceSynchronize();  //cpu waits for gpu completion

    cudaMemcpy(C, d_C, size, cudaMemcpyDeviceToHost);  //copy gpu to cpu

    printf("Result Matrix:\n");
    for(int i=0;i<n;i++){
        for(int j=0;j<n;j++){
            printf("%d ",C[i][j]);
        }
        printf("\n");
    }

    cudaFree(d_A);  //releases gpu memory
    cudaFree(d_B);
    cudaFree(d_C);

    return 0;
}
