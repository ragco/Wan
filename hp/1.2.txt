#include <iostream>   // For input and output
#include <vector>     // For using vector
#include <algorithm>  // For swap function
#include <omp.h>      // OpenMP library for parallel programming

using namespace std;

/* =========================
   SEQUENTIAL BUBBLE SORT
   ========================= */

// Function for Sequential Bubble Sort
// Compares adjacent elements and swaps them
void sequentialBubbleSort(vector<int>& arr) {

    int n = arr.size(); // Size of array

    // Outer loop for passes
    for (int i = 0; i < n - 1; i++) {

        // Inner loop for comparison
        for (int j = 0; j < n - i - 1; j++) {

            // Swap if elements are in wrong order
            if (arr[j] > arr[j + 1]) {

                swap(arr[j], arr[j + 1]);
            }
        }
    }
}

/* =========================
   PARALLEL BUBBLE SORT
   Odd-Even Transposition Sort
   ========================= */

// Function for Parallel Bubble Sort
void parallelBubbleSort(vector<int>& arr) {

    int n = arr.size(); // Size of array

    // Repeat for all passes
    for (int i = 0; i < n; i++) {

        // Even phase
        // Compare even indexed pairs in parallel
        #pragma omp parallel for
        for (int j = 0; j < n - 1; j += 2) {

            if (arr[j] > arr[j + 1]) {

                swap(arr[j], arr[j + 1]);
            }
        }

        // Odd phase
        // Compare odd indexed pairs in parallel
        #pragma omp parallel for
        for (int j = 1; j < n - 1; j += 2) {

            if (arr[j] > arr[j + 1]) {

                swap(arr[j], arr[j + 1]);
            }
        }
    }
}

/* =========================
   MERGE FUNCTION
   ========================= */

// Function to merge two sorted halves
void merge(vector<int>& arr, int left, int mid, int right) {

    int n1 = mid - left + 1; // Size of left subarray

    int n2 = right - mid; // Size of right subarray

    // Temporary arrays
    vector<int> L(n1), R(n2);

    // Copy data into left array
    for (int i = 0; i < n1; i++)
        L[i] = arr[left + i];

    // Copy data into right array
    for (int i = 0; i < n2; i++)
        R[i] = arr[mid + 1 + i];

    int i = 0, j = 0, k = left;

    // Merge arrays in sorted order
    while (i < n1 && j < n2) {

        if (L[i] <= R[j]) {

            arr[k] = L[i];
            i++;

        } else {

            arr[k] = R[j];
            j++;
        }

        k++;
    }

    // Copy remaining elements from left array
    while (i < n1) {

        arr[k] = L[i];
        i++;
        k++;
    }

    // Copy remaining elements from right array
    while (j < n2) {

        arr[k] = R[j];
        j++;
        k++;
    }
}

/* =========================
   SEQUENTIAL MERGE SORT
   ========================= */

// Function for Sequential Merge Sort
void sequentialMergeSort(vector<int>& arr, int left, int right) {

    // Continue if more than one element exists
    if (left < right) {

        // Find middle index
        int mid = left + (right - left) / 2;

        // Sort left half
        sequentialMergeSort(arr, left, mid);

        // Sort right half
        sequentialMergeSort(arr, mid + 1, right);

        // Merge both halves
        merge(arr, left, mid, right);
    }
}

/* =========================
   PARALLEL MERGE SORT
   ========================= */

// Function for Parallel Merge Sort
void parallelMergeSort(vector<int>& arr, int left, int right, int depth) {

    if (left < right) {

        int mid = left + (right - left) / 2;

        // If depth becomes zero use sequential merge sort
        if (depth <= 0) {

            sequentialMergeSort(arr, left, mid);

            sequentialMergeSort(arr, mid + 1, right);

        } else {

            // Create parallel sections
            #pragma omp parallel sections
            {

                // First thread handles left half
                #pragma omp section
                {
                    parallelMergeSort(arr, left, mid, depth - 1);
                }

                // Second thread handles right half
                #pragma omp section
                {
                    parallelMergeSort(arr, mid + 1, right, depth - 1);
                }
            }
        }

        // Merge sorted halves
        merge(arr, left, mid, right);
    }
}

/* =========================
   PRINT ARRAY
   ========================= */

// Function to print array elements
void printArray(vector<int>& arr) {

    for (int x : arr)
        cout << x << " ";

    cout << endl;
}

/* =========================
   MAIN FUNCTION
   ========================= */

int main() {

    int n; // Number of elements

    cout << "Enter number of elements: ";

    cin >> n;

    vector<int> arr(n);

    cout << "Enter elements:\n";

    // Input array elements
    for (int i = 0; i < n; i++) {

        cin >> arr[i];
    }

    // Create copies for different sorting algorithms
    vector<int> bubbleSeq = arr;
    vector<int> bubblePar = arr;
    vector<int> mergeSeq = arr;
    vector<int> mergePar = arr;

    double start, end;

    /* =========================
       SEQUENTIAL BUBBLE SORT
       ========================= */

    start = omp_get_wtime();

    sequentialBubbleSort(bubbleSeq);

    end = omp_get_wtime();

    cout << "\nSequential Bubble Sort:\n";

    printArray(bubbleSeq);

    cout << "Time: " << end - start << " seconds\n";


    /* =========================
       PARALLEL BUBBLE SORT
       ========================= */

    start = omp_get_wtime();

    parallelBubbleSort(bubblePar);

    end = omp_get_wtime();

    cout << "\nParallel Bubble Sort:\n";

    printArray(bubblePar);

    cout << "Time: " << end - start << " seconds\n";


    /* =========================
       SEQUENTIAL MERGE SORT
       ========================= */

    start = omp_get_wtime();

    sequentialMergeSort(mergeSeq, 0, n - 1);

    end = omp_get_wtime();

    cout << "\nSequential Merge Sort:\n";

    printArray(mergeSeq);

    cout << "Time: " << end - start << " seconds\n";


    /* =========================
       PARALLEL MERGE SORT
       ========================= */

    start = omp_get_wtime();

    parallelMergeSort(mergePar, 0, n - 1, 4);

    end = omp_get_wtime();

    cout << "\nParallel Merge Sort:\n";

    printArray(mergePar);

    cout << "Time: " << end - start << " seconds\n";

    return 0;
}
